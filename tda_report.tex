\subsubsection{Α.1—Νέφος Σημείων \& \foreignlanguage{english}{Vietoris--Rips} (\foreignlanguage{english}{from scratch})}
\vspace{0.5em}

\vspace{1em}
\textbf{Εξήγηση Υλοποίησης:}
Για την υλοποίηση εκ του μηδενός, δημιουργήσαμε τα νέφη σημείων και υπολογίσαμε τον πίνακα αποστάσεων. Στη συνέχεια, για μια δεδομένη ακτίνα $r$, βρήκαμε όλες τις ακμές (αποστάσεις $\le 2r$) και όλα τα τρίγωνα (κλίκες μεγέθους 3). Ο υπολογισμός των αριθμών \foreignlanguage{english}{Betti} έγινε μέσω του πίνακα συνόρων (\foreignlanguage{english}{boundary matrix}) και απαλοιφής \foreignlanguage{english}{Gauss}.

\vspace{1em}
\textbf{Σημαντικό Κομμάτι Κώδικα:}
\selectlanguage{english}
\begin{lstlisting}
# Finding edges and triangles (Simplices)
edges = []
for i in range(n):
    for j in range(i+1, n):
        if dist_matrix[i, j] <= 2 * r:
            edges.append((i, j))

triangles = []
for i in range(n):
    for j in range(i+1, n):
        for k in range(j+1, n):
            if dist_matrix[i, j] <= 2*r and dist_matrix[i, k] <= 2*r and dist_matrix[j, k] <= 2*r:
                triangles.append((i, j, k))
\end{lstlisting}
\selectlanguage{greek}
\textit{Σχολιασμός: Ο κώδικας αυτός ελέγχει όλους τους συνδυασμούς σημείων για να βρει αν σχηματίζουν ακμές ή τρίγωνα με βάση την ακτίνα $r$. Είναι η βάση για την κατασκευή του συμπλέγματος.}

\vspace{1em}
\textbf{Αποτελέσματα \& Ερμηνεία:}
Τα αποτελέσματα του υπολογισμού των αριθμών \foreignlanguage{english}{Betti} για τον κύκλο με $\epsilon = 0.5$ έδωσαν $\beta_0 = 1$, $\beta_1 = 1$, $\beta_2 = 1$. \textit{Σημείωση: Η τιμή $\beta_2=1$ οφείλεται στην αποκοπή του συμπλέγματος στη διάσταση 2 (έλλειψη τετραέδρων).}

\foreignlanguage{english}{\includegraphics[width=0.8\textwidth]{point_clouds.png}}
\caption{Τα 4 παραγόμενα νέφη σημείων. Παρατηρούμε τις διαφορετικές τοπολογικές δομές (κύκλος, τόρος, σφαίρα).}
\end{figure}

\subsubsection{Β.1: \foreignlanguage{english}{Persistent Homology} από Νέφη Σημείων}

result = ripser(data, maxdim=2)

# Main logic of fit method in CustomMapper
for int_idx, interval_bounds in enumerate(intervals):
    # Find points in interval
    pts_idx = np.where(in_interval)[0]
    # Cluster subset of points
    labels = self.clustering_algo.fit_predict(X_subset)
    # Add nodes and edges to graph
\end{lstlisting}
\selectlanguage{greek}

\subsubsection{Γ.2: Εφαρμογή \foreignlanguage{english}{Mapper} \& Ανάλυση Ευαισθησίας}
\vspace{0.5em}

\vspace{1em}
\textbf{Αποτελέσματα \& Ερμηνεία:}
Εφαρμόσαμε τον αλγόριθμο στα σύνολα δεδομένων \foreignlanguage{english}{Figure-8} και \foreignlanguage{english}{Horse-shoe} με 3 διαφορετικές συναρτήσεις φίλτρου: \foreignlanguage{english}{PCA} 1ης συνιστώσας, \foreignlanguage{english}{L2-norm} και \foreignlanguage{english}{Density}.

\foreignlanguage{english}{\includegraphics[width=0.32\textwidth]{g2_figure_8_l2-norm_custom.png}}

\textit{Σχολιασμός: Η τριάδα των γραφημάτων για το \foreignlanguage{english}{Figure-8} διαφέρει ανάλογα με το φίλτρο:
1. Το \foreignlanguage{english}{PCA} διαχωρίζει τους δύο κύκλους (αριστερά-δεξιά) στον άξονα μέγιστης διασποράς.
2. Το \foreignlanguage{english}{L2-norm} δείχνει την απόσταση από το κέντρο, δημιουργώντας μια δομή που "ανοίγει" από το κέντρο προς τα άκρα.
3. Το \foreignlanguage{english}{Density} αναδεικνύει το σημείο τομής των δύο κύκλων όπου η πυκνότητα των σημείων είναι μέγιστη.}

\textit{Σχολιασμός: Στο \foreignlanguage{english}{Horse-shoe}:
1. Το \foreignlanguage{english}{PCA} "ξεδιπλώνει" την καμπύλη σε μια συνεχή γραμμική δομή.
2. Το \foreignlanguage{english}{L2-norm} δίνει παρόμοιες τιμές σε όλα τα σημεία (αφού ισαπέχουν από το κέντρο), συμπυκνώνοντας το γράφημα.
3. Το \foreignlanguage{english}{Density} δείχνει ότι η πυκνότητα είναι ομοιόμορφη κατά μήκος του πετάλου, διατηρώντας τη δομή ως μία συνεκτική συνιστώσα.}

\begin{figure}[H]
\centering
\foreignlanguage{english}{\includegraphics[width=0.6\textwidth]{g2_sensitivity_heatmap.png}}
\caption{Ανάλυση ευαισθησίας: Αριθμός συνεκτικών συνιστωσών συναρτήσει των παραμέτρων.}
\end{figure}

\textit{Σχολιασμός: Η ανάλυση ευαισθησίας δείχνει ότι ο αριθμός των συνεκτικών συνιστωσών εξαρτάται έντονα από τον αριθμό των διαστημάτων και το ποσοστό επικάλυψης. Μεγαλύτερη επικάλυψη οδηγεί σε πιο συνεκτικά γραφήματα.}
